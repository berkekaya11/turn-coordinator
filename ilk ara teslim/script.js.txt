function clamp(value, min, max) {
  return Math.min(Math.max(value, min), max);
}

function turnRateToFullTurnTime(rate) {
  const absRate = Math.abs(rate);
  if (absRate < 0.05) return "∞";
  const seconds = 360 / absRate;
  return `${seconds.toFixed(0)} sn`;
}

function calculateBallPosition(turnRate, rudder) {
  const turnEffect = turnRate * 9;
  const rudderEffect = rudder * 0.58;
  const slipSkid = clamp(turnEffect - rudderEffect, -50, 50);
  return 50 + slipSkid * 0.7;
}

function getCoordinationText(ballPercent) {
  if (ballPercent < 42) return "Sola kayma";
  if (ballPercent > 58) return "Sağa kayma";
  return "Merkezde";
}
